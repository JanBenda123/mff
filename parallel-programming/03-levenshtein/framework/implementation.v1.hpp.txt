#ifndef LEVENSHTEIN_IMPLEMENTATION_HPP
#define LEVENSHTEIN_IMPLEMENTATION_HPP

#include <iostream>

template <typename T>
class LevenshteinMatrix
{
public:
	LevenshteinMatrix(size_t rows, size_t cols)
		: rows_(rows), cols_(cols), data_(rows * cols) {}

	T &operator()(size_t row, size_t col)
	{
		return data_[row * cols_ + col];
	}

	void print()
	{
		// return;
		for (std::size_t i = 0; i < (rows_) * (cols_); i++)
		{
			std::size_t col = i % (cols_);
			std::size_t row = i / (cols_);

			if (col == 0)
			{
				std::cout << '\n';
			}
			if (data_[i] >= 10)
			{
				std::cout << data_[i] << ' ';
			}
			else
			{
				std::cout << data_[i] << "  ";
			}
		}
		std::cout << '\n';
	}

private:
	size_t rows_, cols_;
	std::vector<T> data_;
};

template <typename C = char, typename DIST = std::size_t, bool DEBUG = false>
class EditDistance : public IEditDistance<C, DIST, DEBUG>
{
private:
	LevenshteinMatrix<DIST> *lMarix;
	const std::size_t BLOCK_SIZE = 16;

	void compute_block(const std::vector<C> &str1, const std::vector<C> &str2, std::size_t x_start, std::size_t x_end, std::size_t y_start, std::size_t y_end)
	{
		DIST lastLine[BLOCK_SIZE];
		for (std::size_t y = y_start; y < y_end; y++)
		{
			DIST upperLeft = y;
			DIST left = y + 1;
			DIST upper;

			if (x_start != 0)
			{
				left = (*lMarix)(x_start - 1, y);
				upperLeft = (y != 0 ? (*lMarix)(x_start - 1, y - 1) : x_start);
			}

			for (std::size_t x = x_start; x < x_end; x++)
			{
				if (y == 0)
				{ // upper edge of the matrix - generate the values
					upper = x + 1;
				}
				else if (y == y_start)
				{ // first block cycle - needs to load
					upper = (*lMarix)(x, y - 1);
				}
				else
				{ // reuse the local values
					upper = lastLine[x - x_start];
				}

				// calculate value
				DIST dist1 = std::min<DIST>(upper, left) + 1;
				DIST dist2 = upperLeft + (str1[x] == str2[y] ? 0 : 1);
				DIST result = std::min<DIST>(dist1, dist2);
				// save value and shift to right
				lastLine[x - x_start] = result;
				left = result;
				upperLeft = upper;

				if (x == x_end - 1 || y == y_end - 1)
				{ // save only when necessary
					(*lMarix)(x, y) = result;
				}
			}
		}
	}

public:
	/*
	 * \brief Perform the initialization of the functor (e.g., allocate memory buffers).
	 * \param len1, len2 Lengths of first and second string respectively.
	 */
	virtual void init(DIST len1, DIST len2)
	{
		lMarix = new LevenshteinMatrix<DIST>(len1, len2);
	}

	/*
	 * \brief Compute the distance between two strings.
	 * \param str1, str2 Strings to be compared.
	 * \result The computed edit distance.
	 */
	virtual DIST compute(const std::vector<C> &str1, const std::vector<C> &str2)
	{
		std::size_t len1 = str1.size();
		std::size_t len2 = str2.size();

		if (len1 == 0 || len2 == 0)
			return std::max<std::size_t>(len1, len2);

		// Traverse the distanece matrix keeping exactly one row in memory.
		std::size_t blocks_x = len1 / BLOCK_SIZE;
		std::size_t blocks_y = len2 / BLOCK_SIZE;

		// lMarix->print();
#pragma omp parallel
		for (std::size_t i = 0; i < blocks_x + blocks_y - 1; i++)
		{
			std::size_t j_start = (i >= (blocks_y - 1)) ? (i - (blocks_y - 1)) : 0;
			std::size_t j_end = std::min(i + 1, blocks_x);

#pragma omp for
			for (std::size_t j = j_start; j < j_end; j++)
			{
				std::size_t block_x = j;
				std::size_t block_y = i - j;
				compute_block(str1, str2, block_x * BLOCK_SIZE, (block_x + 1) * BLOCK_SIZE, block_y * BLOCK_SIZE, (block_y + 1) * BLOCK_SIZE);
				// lMarix->print();
			}
		}

		// Last item of the last row is the result.
		return (*lMarix)(len1 - 1, len2 - 1);
	}
	~EditDistance()
	{
		if (lMarix != nullptr)
		{
			delete lMarix;
		}
	}
};

#endif
